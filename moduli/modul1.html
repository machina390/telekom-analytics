x = pd.to_numeric(df["godine"], errors="coerce").dropna().to_numpy()

Zadatak 1.4Funkcija za sve mere

≈ 2 h

Napiši funkciju mere(x) koja za numeričku seriju vraća rečnik sa: n, aritmetička sredina, medijana, mod, geometrijska sredina, harmonijska sredina, 10% trimovana sredina, raspon, varijansa uzorka, standardna devijacija, IQR, koeficijent varijacije (%), asimetrija, spljoštenost (višak), Q1, Q3, P95. Primeni je na svih 8 numeričkih kolona sirovih podataka i sačuvaj tabelu rezultati/mere_sirovo.csv.

from scipy import stats

def mere(x: pd.Series) -> dict:
    x = pd.to_numeric(x, errors="coerce").dropna().astype(float)
    poz = x[x > 0]                      # geometrijska i harmonijska traže pozitivne vrednosti
    return {
        "n": len(x),
        "aritmeticka_sredina": x.mean(),
        "medijana": x.median(),
        "mod": x.mode().iloc[0],
        "geometrijska_sredina": stats.gmean(poz),
        "harmonijska_sredina": stats.hmean(poz),
        "trimovana_10": stats.trim_mean(x, 0.10),
        # TODO: raspon, varijansa (ddof=1), std, IQR, koef. varijacije
